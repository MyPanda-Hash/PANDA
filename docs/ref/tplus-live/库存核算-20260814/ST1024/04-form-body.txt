function toggleMoreDropdown(btn) {
  var dropdown = btn.parentElement.querySelector('.tb-group');
  dropdown.style.display = dropdown.style.display === 'none' ? 'block' : 'none';
}
function showMoreDropdown(btn) {
  var dropdown = btn.parentElement.querySelector('.tb-group');
  dropdown.style.display = 'block';
}
function hideMoreDropdown(btn) {
  setTimeout(function() {
    var dropdown = btn.parentElement.querySelector('.tb-group');
    if (!dropdown.matches(':hover') && !btn.matches(':hover')) {
      dropdown.style.display = 'none';
    }
  }, 100);
}
